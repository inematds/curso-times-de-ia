entry_price = 150.00 stop_loss = 142.00 risk_per_share = entry_price - stop_loss # $8.00

account_sizes = [10000, 25000, 50000, 100000, 250000] risk_percentages = [0.01, 0.02, 0.03]

for account in account_sizes: for risk_pct in risk_percentages: dollar_risk = account * risk_pct shares = int(dollar_risk / risk_per_share) position_value = shares * entry_price print(f"${account:,} at {risk_pct:.0%}: {shares} shares (${position_value:,.0f})")
